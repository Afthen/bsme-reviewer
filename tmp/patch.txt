/**
 * scripts/lib/ctx_cache.js
 * Metadata-only ledger of source lines and computed outputs delivered to a model.
 * Source text is never written to the ledger.
 */

import fs from "node:fs";
import path from "node:path";
import crypto from "node:crypto";

const CACHE_RELATIVE = path.join("tmp", "ctx_cache.txt");
const HEADER = "# ctx-cache-v1";
const MAX_LINE_RECORDS = 20000;
const MAX_OUTPUT_RECORDS = 256;

export function hashText(text) {
  return crypto.createHash("sha256").update(String(text), "utf8").digest("hex");
}

function lineKey(rel, line) {
  return `${rel}\0${line}`;
}

function trimOldest(map, limit) {
  if (map.size <= limit) return;
  const rows = [...map.entries()].sort((a, b) => a[1].seq - b[1].seq);
  for (const [key] of rows.slice(0, map.size - limit)) map.delete(key);
}

export function loadContextCache(root = process.cwd()) {
  const file = path.resolve(root, CACHE_RELATIVE);
  const cache = {
    file,
    nextSeq: 1,
    lines: new Map(),
    outputs: new Map(),
    dirty: false
  };

  if (!fs.existsSync(file)) return cache;

  const rows = fs.readFileSync(file, "utf8").split(/\r?\n/);
  if (rows[0] !== HEADER) return cache;

  for (const row of rows.slice(1)) {
    if (!row) continue;
    const fields = row.split("\t");

    try {
      if (fields[0] === "L" && fields.length === 7) {
        const seq = Number(fields[1]);
        const rel = Buffer.from(fields[2], "base64url").toString("utf8");
        const line = Number(fields[3]);
        const contentHash = fields[4];
        const complete = fields[5] === "1";
        const displayHash = fields[6];

        if (!Number.isSafeInteger(seq) || seq < 1 ||
            !Number.isSafeInteger(line) || line < 1 ||
            !rel || !/^[a-f0-9]{64}$/.test(contentHash) ||
            !/^[a-f0-9]{64}$/.test(displayHash)) continue;

        cache.lines.set(lineKey(rel, line), {
          seq, rel, line, contentHash, complete, displayHash
        });
        cache.nextSeq = Math.max(cache.nextSeq, seq + 1);
      } else if (fields[0] === "O" && fields.length === 3) {
        const seq = Number(fields[1]);
        const outputHash = fields[2];

        if (!Number.isSafeInteger(seq) || seq < 1 ||
            !/^[a-f0-9]{64}$/.test(outputHash)) continue;

        cache.outputs.set(outputHash, { seq, outputHash });
        cache.nextSeq = Math.max(cache.nextSeq, seq + 1);
      }
    } catch {
      // Ignore malformed rows; valid rows remain usable.
    }
  }

  trimOldest(cache.lines, MAX_LINE_RECORDS);
  trimOldest(cache.outputs, MAX_OUTPUT_RECORDS);
  return cache;
}

export function allocateCacheSequence(cache) {
  const seq = cache.nextSeq++;
  cache.dirty = true;
  return seq;
}

/**
 * A complete READ satisfies later READs and GREP excerpts.
 * A partial GREP excerpt only satisfies the same displayed excerpt; it must
 * never suppress a later complete READ of that line.
 */
export function findLineDelivery(cache, ref) {
  const rel = String(ref.rel).replace(/\\/g, "/");
  const previous = cache.lines.get(lineKey(rel, ref.line));
  if (!previous || previous.contentHash !== hashText(ref.raw)) return null;
  if (previous.complete) return previous;

  if (!ref.complete && previous.displayHash === hashText(ref.display)) {
    return previous;
  }
  return null;
}

export function recordLineDelivery(cache, ref, seq) {
  const rel = String(ref.rel).replace(/\\/g, "/");
  cache.lines.set(lineKey(rel, ref.line), {
    seq,
    rel,
    line: ref.line,
    contentHash: hashText(ref.raw),
    complete: Boolean(ref.complete),
    displayHash: hashText(ref.display)
  });
  cache.dirty = true;
  trimOldest(cache.lines, MAX_LINE_RECORDS);
}

export function findOutputDelivery(cache, outputHash) {
  return cache.outputs.get(outputHash) || null;
}

export function recordOutputDelivery(cache, outputHash, seq) {
  cache.outputs.set(outputHash, { seq, outputHash });
  cache.dirty = true;
  trimOldest(cache.outputs, MAX_OUTPUT_RECORDS);
}

export function saveContextCache(cache) {
  if (!cache.dirty) return;

  const rows = [HEADER];
  const lineRows = [...cache.lines.values()].sort((a, b) => a.seq - b.seq);
  const outputRows = [...cache.outputs.values()].sort((a, b) => a.seq - b.seq);

  for (const item of lineRows) {
    const encodedPath = Buffer.from(item.rel, "utf8").toString("base64url");
    rows.push([
      "L",
      item.seq,
      encodedPath,
      item.line,
      item.contentHash,
      item.complete ? "1" : "0",
      item.displayHash
    ].join("\t"));
  }

  for (const item of outputRows) {
    rows.push(["O", item.seq, item.outputHash].join("\t"));
  }

  fs.mkdirSync(path.dirname(cache.file), { recursive: true });
  const temp = `${cache.file}.tmp-${process.pid}`;
  fs.writeFileSync(temp, `${rows.join("\n")}\n`, "utf8");
  fs.renameSync(temp, cache.file);
  cache.dirty = false;
}

export function resetContextCache(root = process.cwd()) {
  const dir = path.resolve(root, "tmp");
  const file = path.join(dir, "ctx_cache.txt");

  if (fs.existsSync(file)) fs.rmSync(file, { force: true });
  if (!fs.existsSync(dir)) return;

  for (const name of fs.readdirSync(dir)) {
    if (name.startsWith("ctx_cache.txt.tmp-")) {
      fs.rmSync(path.join(dir, name), { force: true });
    }
  }
}

function analyzeSourceResult(res, cache) {
  const refs = res.sourceRefs || [];
  if (!refs.length) {
    return { res, cachedCount: 0, markerCount: 0, newRefs: [] };
  }

  const cached = [];
  const newRefs = [];
  const seen = new Set();

  for (const ref of refs) {
    const key = `${ref.rel}\0${ref.line}`;
    const previous = findLineDelivery(cache, ref);

    if (previous) {
      cached.push({ ref, seq: previous.seq });
    } else if (seen.has(key)) {
      cached.push({ ref, seq: null });
    } else {
      seen.add(key);
      newRefs.push(ref);
    }
  }

  if (!cached.length) {
    return {
      res: { ...res, sourceRefs: newRefs },
      cachedCount: 0,
      markerCount: 0,
      newRefs
    };
  }

  const lines = res.text.split("\n");
  const groups = new Map();
  for (const item of cached) {
    const group = item.ref.group ?? 0;
    if (!groups.has(group)) groups.set(group, []);
    groups.get(group).push(item);
  }

  for (const [group, items] of groups) {
    items.sort((a, b) => a.ref.outputIndex - b.ref.outputIndex);
    const indexes = items.map((item) => item.ref.outputIndex);
    const deliveries = [...new Set(items.map((item) => item.seq).filter(Number.isInteger))];
    const deliveryText = deliveries.length ? ` (earlier delivery${deliveries.length > 1 ? "ies" : ""}: ${deliveries.join(", ")})` : "";
    const first = indexes[0];

    lines[first] =
      `[CACHED: ${items.length} source line(s) already included${deliveryText}]`;
    for (const index of indexes.slice(1)) lines[index] = "";
  }

  return {
    res: {
      ...res,
      text: lines.join("\n"),
      lines: Math.max(1, res.lines - cached.length + groups.size),
      sourceRefs: newRefs
    },
    cachedCount: cached.length,
    markerCount: groups.size,
    newRefs
  };
}

/**
 * Prepares a response and a deferred ledger commit.
 * The caller commits only after its response file has been written successfully.
 */
export async function prepareFulfillment(raw, {
  root = process.cwd(),
  cache: useCache = false
} = {}) {
  const directives = parseDirectives(raw);
  if (directives.length === 0) return { text: "", commit() {} };

  const cache = useCache ? loadContextCache(root) : null;
  const run = directives.slice(0, LIMITS.MAX_DIRECTIVES);
  const dropped = directives.slice(LIMITS.MAX_DIRECTIVES);
  const parts = [];
  let used = 0;
  const DIRECTIVE_RE = /^(READ|OUTLINE|GREP|TRACE|GRAPH|TOPIC|UNIT|CONVERT|SOLVE|HYGIENE|FAILURES|DERIVE|AUDIT-LATEX|AUDIT-STEP|RENDER-STEP|PROBLEM-STEPS|STEP-AST|DIFF-STEP|DIFF-DERIVE|ROUNDTRIP-STEP)(?:\s+(.*))?$/;

  for (const line of run) {
    const remaining = Math.max(0, LIMITS.MAX_OUTPUT_LINES - used);
    const m = line.match(DIRECTIVE_RE);
    let r;

    if (m) {
      const verb = m[1];
      const arg = (m[2] || "").trim();

      try {
        if (cache && (verb === "READ" || verb === "GREP")) {
          let res = HANDLERS[verb](arg, { root, remaining: Number.MAX_SAFE_INTEGER });
          let analyzed = analyzeSourceResult(res, cache);

          if (analyzed.res.lines > remaining && (res.sourceRefs || []).length) {
            const overhead = Math.max(0, res.lines - res.sourceRefs.length);
            const retryRemaining = Math.max(
              0,
              remaining + analyzed.cachedCount - analyzed.markerCount - overhead
            );
            res = HANDLERS[verb](arg, { root, remaining: retryRemaining });
            analyzed = analyzeSourceResult(res, cache);
          }

          r = analyzed.res;

          if (analyzed.newRefs.length) {
            const seq = allocateCacheSequence(cache);
            for (const ref of analyzed.newRefs) {
              recordLineDelivery(cache, ref, seq);
            }
          }
        } else {
          const result = HANDLERS[verb](arg, { root, remaining });
          r = result instanceof Promise ? await result : result;

          if (cache && verb !== "READ" && verb !== "GREP" &&
              r.cacheable !== false && r.text) {
            const outputHash = hashText(r.text);
            const previous = findOutputDelivery(cache, outputHash);

            if (previous) {
              r = {
                text: `[CACHED: identical ${verb} result already delivered in context delivery #${previous.seq}]\n`,
                lines: 1
              };
            } else {
              const seq = allocateCacheSequence(cache);
              recordOutputDelivery(cache, outputHash, seq);
            }
          }
        }
      } catch (err) {
        r = { text: `[${verb} ERROR: ${err.message}]\n`, lines: 1, cacheable: false };
      }
    } else {
      r = { text: `[UNRECOGNIZED: ${line}]\n${USAGE}\n`, lines: 1, cacheable: false };
    }

    parts.push(r.text);
    used += r.lines;
  }

  let header = `BUDGET: ${used}/${LIMITS.MAX_OUTPUT_LINES} output lines, ${run.length}/${LIMITS.MAX_DIRECTIVES} directives used.`;
  if (dropped.length) {
    header += `\nNOT RUN (over the ${LIMITS.MAX_DIRECTIVES}-directive limit; prioritize and resend):\n${dropped.join("\n")}`;
  }

  return {
    text: `${header}\n\n${parts.join("\n")}`,
    commit() {
      if (cache) saveContextCache(cache);
    }
  };
}

/** Fulfills a raw request; caching is opt-in for programmatic callers. */
export async function fulfill(raw, options = {}) {
  const prepared = await prepareFulfillment(raw, options);
  prepared.commit();
  return prepared.text;
}

    const first = await runCached("READ lib.js#K");

    assert.doesNotMatch(second, /4: line 4/);
    assert.doesNotMatch(second, /5: line 5/);
    assert.match(second, /6: line 6/);
    assert.match(second, /8: line 8/);
