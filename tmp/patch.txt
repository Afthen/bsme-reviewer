### Root Cause Analysis

1. **Static Manifest vs. Live Loaded Data:**
   - In `app.js`, `setupMcqSelector()` populated the dropdown labels using static `m.count` from `MCQ_MANIFEST` (`${m.title} (${m.count})`).
   - In `mcq_view.js`, the main setup subtitle calculates `totalAvail` dynamically by inspecting `s.questions.length` of the actual loaded sets (`412 questions`).
   - If `src/mcq/index.js` was cached by the browser or loaded before the expanded question sets, the dropdown showed the stale static counts (`15, 25, 14...`) while the lobby showed the true count (`412`).

2. **Resolution:**
   - Maintain a live counts map in `app.js`.
   - In `refreshMcqMode()`, immediately sync each loaded set's real `questions.length` into the dropdown labels so the dropdown and the setup card stay in 100% lockstep parity.

async function refreshMcqMode() {
  const sets = await getSelectedMcqSets();
  for (const s of sets) {
    const qCount = s.questions?.length ?? s.count ?? 0;
    liveMcqCounts.set(s.id, qCount);
    const leaf = document.querySelector(`.topic-leaf[data-mcq-id="${s.id}"] .topic-leaf-title`);
    if (leaf) {
      leaf.textContent = `${s.title} (${qCount})`;
    }
  }
  const view = document.getElementById("view-mcq");
  if (!view) return;
  if (sets.length === 0) {
    renderMcqEmptyPrompt();
  } else {
    view.classList.remove("is-empty");
    renderMcqSets(sets);
  }
  updateMcqSidebar(sets);
}
