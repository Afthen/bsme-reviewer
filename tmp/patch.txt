/**
 * src/core/relation_builder.js
 * Unified Relation Constructor & Continued-Formula Compiler.
 * Supports standard equations, SI/US variants, requiredUnits, multi-"=" chains,
 * indexed (_i) laws, and nested continued operators: sum(...), prod(...), chain(...).
 */

import { parseMathAST, astToExprStr, getAstIdentifiers, unwrapGroup } from "../cas/ast.js";
import { isReservedMathToken } from "../cas/algebra.js";
import { astToLatex } from "../cas/latex.js";

const DEFAULT_COUNT = 3;

function instantiateIdName(name, idx) {
  if (name.endsWith("_i")) {
    return `${name.slice(0, -1)}${idx}`;
  }
  return name;
}

function instantiateAST(node, idx) {
  if (!node) return null;
  if (node.type === "id") return { type: "id", name: instantiateIdName(node.name, idx) };
  if (node.type === "group") return { type: "group", expr: instantiateAST(node.expr, idx) };
  if (node.type === "call") return { type: "call", fn: node.fn, arg: instantiateAST(node.arg, idx) };
  if (node.type === "binary") {
    return {
      type: "binary",
      op: node.op,
      left: instantiateAST(node.left, idx),
      right: instantiateAST(node.right, idx)
    };
  }
  return { ...node };
}

function astHasIndexedToken(node) {
  return getAstIdentifiers(node).some(id => id.endsWith("_i"));
}

function astHasFoldCall(node) {
  if (!node) return false;
  if (node.type === "call") {
    if (node.fn === "sum" || node.fn === "prod" || node.fn === "chain") return true;
    return astHasFoldCall(node.arg);
  }
  if (node.type === "group") return astHasFoldCall(node.expr);
  if (node.type === "binary") return astHasFoldCall(node.left) || astHasFoldCall(node.right);
  return false;
}

function expandFoldCallsAST(node, count) {
  if (!node) return null;
  if (node.type === "group") return { type: "group", expr: expandFoldCallsAST(node.expr, count) };
  if (node.type === "call") {
    if (node.fn === "sum" || node.fn === "prod") {
      const op = node.fn === "sum" ? "+" : "*";
      const terms = [];
      for (let i = 1; i <= count; i++) {
        let termAst = expandFoldCallsAST(instantiateAST(node.arg, i), count);
        if (op === "*" && termAst.type === "binary" && (termAst.op === "+" || termAst.op === "-")) {
          termAst = { type: "group", expr: termAst };
        }
        terms.push(termAst);
      }
      let acc = terms[0];
      for (let i = 1; i < terms.length; i++) {
        acc = { type: "binary", op, left: acc, right: terms[i] };
      }
      return { type: "group", expr: acc };
    }
    return { type: "call", fn: node.fn, arg: expandFoldCallsAST(node.arg, count) };
  }
  if (node.type === "binary") {
    return {
      type: "binary",
      op: node.op,
      left: expandFoldCallsAST(node.left, count),
      right: expandFoldCallsAST(node.right, count)
    };
  }
  return node;
}

function astToEllipsisLatex(node, varsDb = {}) {
  if (!node) return "";
  if (node.type === "group") {
    const inner = unwrapGroup(node.expr);
    if (inner?.type === "call" && (inner.fn === "sum" || inner.fn === "prod")) {
      return `\\left(${astToEllipsisLatex(inner, varsDb)}\\right)`;
    }
    return `(${astToEllipsisLatex(node.expr, varsDb)})`;
  }
  if (node.type === "call") {
    if (node.fn === "sum" || node.fn === "prod") {
      const t1 = astToEllipsisLatex(instantiateAST(node.arg, 1), varsDb);
      const t2 = astToEllipsisLatex(instantiateAST(node.arg, 2), varsDb);
      const tn = astToEllipsisLatex(instantiateAST(node.arg, "n"), varsDb);
      if (node.fn === "sum") return `${t1} + ${t2} + \\dots + ${tn}`;
      return `${t1} \\cdot ${t2} \\cdots ${tn}`;
    }
    if (node.fn === "chain") {
      const t1 = astToEllipsisLatex(instantiateAST(node.arg, 1), varsDb);
      const t2 = astToEllipsisLatex(instantiateAST(node.arg, 2), varsDb);
      const tn = astToEllipsisLatex(instantiateAST(node.arg, "n"), varsDb);
      return `${t1} = ${t2} = \\dots = ${tn}`;
    }
    const wrapper = { ...node, arg: astToEllipsisLatexNodeSafe(node.arg, varsDb) };
    if (typeof wrapper.arg === "string") return `\\text{${node.fn}}\\left(${wrapper.arg}\\right)`;
    return astToLatex(node, null, varsDb);
  }
  if (node.type === "binary") {
    if (node.op === "/") {
      return `\\cfrac{${astToEllipsisLatex(node.left, varsDb)}}{${astToEllipsisLatex(node.right, varsDb)}}`;
    }
    if (node.op === "*") {
      const L = astToEllipsisLatex(node.left, varsDb);
      const R = astToEllipsisLatex(node.right, varsDb);
      const uR = unwrapGroup(node.right);
      const wrapR = (uR?.type === "call" && uR.fn === "sum") ? `\\left(${R}\\right)` : R;
      return `${L} \\, ${wrapR}`;
    }
    if (node.op === "+" || node.op === "-") {
      return `${astToEllipsisLatex(node.left, varsDb)} ${node.op} ${astToEllipsisLatex(node.right, varsDb)}`;
    }
    if (node.op === "^") {
      return `{${astToEllipsisLatex(node.left, varsDb)}}^{${astToEllipsisLatex(node.right, varsDb)}}`;
    }
  }
  return astToLatex(node, null, varsDb);
}

function astToEllipsisLatexNodeSafe(node, varsDb) {
  if (!astHasFoldCall(node)) return node;
  return astToEllipsisLatex(node, varsDb);
}

function extractNodesFromFormula(formulaStr) {
  const parts = formulaStr.split("=").map(s => s.trim());
  const seen = new Set();
  const ordered = [];
  for (const part of parts) {
    const ast = parseMathAST(part);
    if (!ast) continue;
    for (const id of getAstIdentifiers(ast)) {
      if (!isReservedMathToken(id) && !seen.has(id)) {
        seen.add(id);
        ordered.push(id);
      }
    }
  }
  return ordered;
}

function compileSingleEquation(spec, formulaStr, count, varsDb, idOverride = null, masteryIdOverride = null, texOverride = null) {
  const parts = formulaStr.split("=").map(s => s.trim());
  const lhsAst = parseMathAST(parts[0]);
  const rhsAst = parseMathAST(parts[1]);

  if (!lhsAst || !rhsAst) {
    throw new Error(`Invalid formula in relation [${spec.id}]: "${formulaStr}"`);
  }

  // Case 1: Symmetric equality chain: lhs = chain(term_i)
  const uRhs = unwrapGroup(rhsAst);
  if (uRhs?.type === "call" && uRhs.fn === "chain") {
    const lhsExpanded = expandFoldCallsAST(lhsAst, count);
    const lhsStr = astToExprStr(lhsExpanded);
    const lhsTex = astToEllipsisLatex(lhsAst, varsDb);
    const chainTex = astToEllipsisLatex(uRhs, varsDb);
    const canonicalTex = texOverride || `${lhsTex} = ${chainTex}`;

    const out = [];
    for (let i = 1; i <= count; i++) {
      const termInst = expandFoldCallsAST(instantiateAST(uRhs.arg, i), count);
      const baseFormula = `${astToExprStr(termInst)} = ${lhsStr}`;
      out.push({
        id: `${spec.id}__${i}`,
        masteryId: spec.id,
        name: spec.name,
        canonicalTex,
        baseFormula,
        nodes: extractNodesFromFormula(baseFormula),
        ...(spec.requiredUnits ? { requiredUnits: spec.requiredUnits } : {})
      });
    }
    return out;
  }

  // Case 2: Indexed per-element equation outside sum/prod (e.g. delta_i = F_T / k_i)
  const lhsHasFold = astHasFoldCall(lhsAst);
  const rhsHasFold = astHasFoldCall(rhsAst);
  const templateWithoutFolds = {
    lhs: lhsHasFold ? null : lhsAst,
    rhs: rhsHasFold ? null : rhsAst
  };
  const hasLooseIndexed =
    (!lhsHasFold && astHasIndexedToken(lhsAst)) ||
    (!rhsHasFold && astHasIndexedToken(rhsAst));

  if (hasLooseIndexed) {
    const canonicalTex = texOverride || `${astToEllipsisLatex(lhsAst, varsDb)} = ${astToEllipsisLatex(rhsAst, varsDb)}`;
    const out = [];
    for (let i = 1; i <= count; i++) {
      const lInst = expandFoldCallsAST(instantiateAST(lhsAst, i), count);
      const rInst = expandFoldCallsAST(instantiateAST(rhsAst, i), count);
      const baseFormula = `${astToExprStr(lInst)} = ${astToExprStr(rInst)}`;
      out.push({
        id: `${spec.id}__${i}`,
        masteryId: spec.id,
        name: spec.name,
        canonicalTex,
        baseFormula,
        nodes: extractNodesFromFormula(baseFormula),
        ...(spec.requiredUnits ? { requiredUnits: spec.requiredUnits } : {})
      });
    }
    return out;
  }

  // Case 3: Standard or folded (sum/prod) binary equation
  void templateWithoutFolds;
  const expandedLhs = expandFoldCallsAST(lhsAst, count);
  const expandedRhs = expandFoldCallsAST(rhsAst, count);
  const baseFormula = `${astToExprStr(expandedLhs)} = ${astToExprStr(expandedRhs)}`;
  const canonicalTex = texOverride || `${astToEllipsisLatex(lhsAst, varsDb)} = ${astToEllipsisLatex(rhsAst, varsDb)}`;

  return [{
    id: idOverride || spec.id,
    masteryId: masteryIdOverride || spec.id,
    name: spec.name,
    canonicalTex,
    baseFormula,
    nodes: extractNodesFromFormula(baseFormula),
    ...(spec.requiredUnits ? { requiredUnits: spec.requiredUnits } : {})
  }];
}

function compileVariantMap(variantsSpec, count, varsDb, spec) {
  if (!variantsSpec) return null;
  const compiled = {};
  for (const [sysKey, vSpec] of Object.entries(variantsSpec)) {
    const fStr = vSpec.formula || vSpec.baseFormula;
    const [res] = compileSingleEquation(
      spec,
      fStr,
      count,
      varsDb,
      spec.id,
      spec.id,
      vSpec.canonicalTex || null
    );
    compiled[sysKey] = {
      canonicalTex: res.canonicalTex,
      baseFormula: res.baseFormula,
      nodes: res.nodes,
      ...(vSpec.requiredUnits ? { requiredUnits: vSpec.requiredUnits } : {})
    };
  }
  return compiled;
}

export function defineRelation(spec, options = {}) {
  const count = options.count ?? spec.count ?? DEFAULT_COUNT;
  const varsDb = options.varsDb || {};
  const rawFormula = spec.formula || spec.baseFormula;

  if (!rawFormula || !rawFormula.includes("=")) {
    throw new Error(`Relation [${spec.id}] missing valid "=" formula`);
  }

  const segments = rawFormula.split("=").map(s => s.trim()).filter(Boolean);
  const compiledVariants = compileVariantMap(spec.variants, count, varsDb, spec);

  // Multi-"=" chain: A = B = C -> distinct binary edges A = B, A = C
  if (segments.length > 2) {
    const chainTex = spec.canonicalTex || segments
      .map(seg => astToEllipsisLatex(parseMathAST(seg), varsDb))
      .join(" = ");

    const out = [];
    for (let j = 1; j < segments.length; j++) {
      const subFormula = `${segments[0]} = ${segments[j]}`;
      const subId = `${spec.id}__${j}`;
      const entries = compileSingleEquation(spec, subFormula, count, varsDb, subId, subId, chainTex);
      entries.forEach(e => {
        if (compiledVariants) e.variants = compiledVariants;
        out.push(e);
      });
    }
    return out;
  }

  const entries = compileSingleEquation(spec, rawFormula, count, varsDb, spec.id, spec.id, spec.canonicalTex || null);
  entries.forEach(e => {
    if (compiledVariants) e.variants = compiledVariants;
  });
  return entries.length === 1 ? entries[0] : entries;
}

export function compileRelations(items = [], options = {}) {
  const flat = [];
  for (const item of items) {
    if (!item) continue;
    if (Array.isArray(item)) {
      flat.push(...compileRelations(item, options));
      continue;
    }
    // Already-compiled concrete relation (e.g. spread from a common/ base file)
    if (item.baseFormula && Array.isArray(item.nodes) && !item.formula) {
      flat.push(item);
      continue;
    }
    const built = defineRelation(item, options);
    if (Array.isArray(built)) flat.push(...built);
    else flat.push(built);
  }
  return Object.freeze(flat);
}

export function expandIndexedVarsDb(templateDb, count = DEFAULT_COUNT) {
  const out = {};
  const replaceIdx = (str, i) => typeof str === "string"
    ? str.replaceAll("{i}", String(i)).replaceAll("_i", `_${i}`)
    : str;

  for (const [key, meta] of Object.entries(templateDb)) {
    if (!key.endsWith("_i")) {
      out[key] = meta;
      continue;
    }
    const stem = key.slice(0, -1); // e.g. "k_"
    for (let i = 1; i <= count; i++) {
      const concreteKey = `${stem}${i}`;
      out[concreteKey] = {
        ...meta,
        name: replaceIdx(meta.name, i),
        symbol: replaceIdx(meta.symbol, i),
        ...(meta.aliases ? { aliases: meta.aliases.map(a => replaceIdx(a, i)) } : {}),
        ...(meta.nameAliases ? { nameAliases: meta.nameAliases.map(a => replaceIdx(a, i)) } : {}),
        ...(meta.symbolAliases ? { symbolAliases: meta.symbolAliases.map(a => replaceIdx(a, i)) } : {})
      };
    }
  }
  return Object.freeze(out);
}

 * migrate_relations.mjs
 * Temporary deterministic codemod: converts all existing relations.js files under
 * src/topics/ to use defineRelation + compileRelations (drops manual nodes arrays,

import fs from "node:fs";
import path from "node:path";
import vm from "node:vm";

const TOPICS_DIR = path.resolve("src/topics");
const BUILDER_PATH = path.resolve("src/core/relation_builder.js");

function findRelationFiles(dir) {
  const results = [];
  for (const entry of fs.readdirSync(dir, { withFileTypes: true })) {
    const full = path.join(dir, entry.name);
    if (entry.isDirectory()) {
      if (entry.name === "springs") continue;
      results.push(...findRelationFiles(full));
    } else if (entry.name.endsWith("relations.js")) {
      results.push(full);
    }
  }
  return results;
}

function splitTopLevelArrayItems(arrBody) {
  const items = [];
  let depth = 0;
  let inStr = null;
  let inLineComment = false;
  let inBlockComment = false;
  let start = 0;

  for (let i = 0; i < arrBody.length; i++) {
    const ch = arrBody[i];
    const next = arrBody[i + 1];

    if (inLineComment) {
      if (ch === "\n") inLineComment = false;
      continue;
    }
    if (inBlockComment) {
      if (ch === "*" && next === "/") { inBlockComment = false; i++; }
      continue;
    }
    if (inStr) {
      if (ch === "\\") { i++; continue; }
      if (ch === inStr) inStr = null;
      continue;
    }
    if (ch === "/" && next === "/") { inLineComment = true; i++; continue; }
    if (ch === "/" && next === "*") { inBlockComment = true; i++; continue; }
    if (ch === `"` || ch === `'` || ch === "`") { inStr = ch; continue; }

    if (ch === "{" || ch === "[" || ch === "(") depth++;
    else if (ch === "}" || ch === "]" || ch === ")") depth--;
    else if (ch === "," && depth === 0) {
      const slice = arrBody.slice(start, i).trim();
      if (slice) items.push(slice);
      start = i + 1;
    }
  }
  const tail = arrBody.slice(start).trim();
  if (tail) items.push(tail);
  return items;
}

function formatSpecObj(obj) {
  const out = {
    id: obj.id,
    name: obj.name,
    formula: obj.baseFormula || obj.formula
  };
  if (obj.canonicalTex) out.canonicalTex = obj.canonicalTex;
  if (obj.requiredUnits) out.requiredUnits = obj.requiredUnits;
  if (obj.variants) {
    out.variants = {};
    for (const [k, v] of Object.entries(obj.variants)) {
      const vOut = { formula: v.baseFormula || v.formula };
      if (v.canonicalTex) vOut.canonicalTex = v.canonicalTex;
      if (v.requiredUnits) vOut.requiredUnits = v.requiredUnits;
      out.variants[k] = vOut;
    }
  }
  return `defineRelation(${JSON.stringify(out, null, 2).replace(/\n/g, "\n  ")})`;
}

  if (!exportMatch) {

  const openBracketIdx = raw.indexOf("[", exportMatch.index);

## 1. Engineering Philosophy: Root Causes over Symptoms

Software engineering in this repository is governed by physical and mathematical principles, not ad-hoc heuristics:

1. **Fix Root Causes, Never Patch Symptoms (Anti-Whack-a-Mole):**
   When an unexpected value appears, identify the mathematical breakdown at the foundational layer. Never apply per-variable scalar patches, hardcoded string comparison branches, or localized `if (key === '...')` overrides.
2. **Design for General Invariants (Branchless Mathematics):**
   Prefer abstract mathematical representations over nested `if` statements. Units are affine metric spaces $(s, o)$ with 7D dimensional vectors. Scale parity, companion resolutions, continued formulas (`...`), and factor-label decompositions must fall out naturally from continuous AST equations, not system-specific switches.
3. **Avoid Tunnel Vision:**
   Never evaluate a formula in isolation. Ask: *How does this change affect the solution DAG? How does it behave in a MIXED unit problem? Does the derivation reflect the actual arithmetic performed? Will ancestor equations receive reinforcement?*
4. **Anti-Pattern Blacklist:**
   - **No string comparisons for physical scale:** Never do `if (u === "°F")` or `if (sys === "US")` to decide absolute temperature targets. Use affine space definitions.
   - **No hardcoded conversion constants or rounded duplicates:** Never hardcode `2.205`, `3,412.14`, or bespoke test scale arrays. `definitions.js` and `scales.js` are the Single Source of Truth.
   - **No per-variable conversion logic:** All unit conversions route through the semantic unit AST and dimensional registry.
   - **No regex-based syntax assertions on math strings:** Tests must validate structural ASTs, balance, and KaTeX rendering. Do not write catch-all regexes like `!/,/.test()` that falsely reject valid math subscripts like $c_{p,w}$ or $m_{m,w}$.
   - **No fragmentation of coupled state points:** Thermodynamic states of the same working fluid must never be split across incompatible units.
   - **No bespoke solvers:** If an equation cannot be solved by `solveDAG` and `isolateVariable`, fix the algebraic graph, never write a hardcoded solver.

## 2. Hard Rules (Non-Negotiable)

### Math and Formulas
- **R1. Formulas are nodes.** Equations are declared via the unified relation constructor (`defineRelation` / `compileRelations`); variables are edges derived automatically from the AST. `solveDAG` finds the path dynamically. Never write bespoke "solve for X" functions or hardcoded solution orders.
- **R2. AST only.** All symbolic operations pass through `src/cas/ast.js`, `src/cas/algebra.js`, and `src/core/relation_builder.js`. No regex rewriting of formulas, no `eval` / `new Function`, no runtime CAS dependencies.
- **R3. Obey formula language rules (section 5).** Explicit `*`, parenthesised complex exponents, binary AST equality (`=` per edge; multi-`=` and `chain(...)` expand automatically into binary edges). Unary `-`, right-associative `^`, arbitrary angle expressions inside trig functions, and nested `sum(...)` / `prod(...)` / `chain(...)` continued operators are natively supported.
- **R4. Case-sensitive canonical keys.** Variable keys match `[A-Za-z][A-Za-z0-9_]*` (`P_d`, `D_P1`, `m_cw`). Never lowercase or normalize keys. LaTeX formatting belongs in `symbol`, never in the canonical key.

### Physical Units & Dimensions
- **R5. Relations are pure physics.** No unit-conversion constants (`25.4`, `1000`, `60`, `3600`, `12`, `1.341`). Coherent base conversions handle scaling automatically.
- **R6. Declare empirical exceptions.**
  - Empirical correlations valid only in specific units: declare `requiredUnits: { token: "unit", targetVar: "unit" }`.
  - Differing mathematical formulations between SI and US: declare `variants.SI` and `variants.US`.
- **R7. Register every unit.** Every unit string must exist in `src/cas/units/definitions.js`, with dimensions and conversion scales registered across `dimensions.js`, `scales.js`, and `unit_sampler.js`. Unknown units fail validation.
- **R8. Thermodynamic Temperature Consistency.**
  - Ideal-gas and isentropic relations requiring absolute temperatures ($T_2 = T_1 r_k^{k-1}$) strictly evaluate in absolute scales ($\text{K}$ in SI, $^\circ\text{R}$ in US).
  - All state point temperatures of a cycle share an identical unit when appearing as deliverables.
  - Affine conversions must use exact offsets ($\pm 273.15$, $\pm 459.67$) and map strictly to their natural absolute companion ($^\circ\text{F} \to ^\circ\text{R}$, $^\circ\text{C} \to \text{K}$).

### Topics, Fractal Architecture, and Data Contracts
- **R9. Ground truth is independent.** `system.js` computes true values by hand with explicit arithmetic. It never calls the solver or unit engine.
- **R10. `createSystem` contract.** `createSystem({ sys })` accepts `"SI"` or `"US"`, returns `sys`, `_u` (unit map), and `_names`, and provides values for every non-abstract variable in `varsDb`. SI-only topics ignore `sys` but still return `sys: "SI"`. Variadic continued-formula topics default `n` deterministically when `n` is omitted so static validation stays exact.
- **R11. Globally unique equation IDs** (`eq_<topic>_<name>`), with expanded indexed/continued edges sharing one `masteryId` when they express the same symmetric law across $i = 1 \dots n$.
- **R12. Fractal composition: share, don't copy.** Folders repeat a self-similar fractal structure: whenever two or more child topics share schemas or relations, extract a `common/` folder at that branch (`index.js`, `schema.js`, `relations.js`) and re-export through `index.js` at every level. Shared equations use the `eq_base_` prefix, and children always import from `../common/index.js`.
- **R13. Clean schemas.** Zero orphan variables, zero duplicate formulas within a topic, and zero alias collisions across `varsDb` (including indexed expansions `k_1..k_n`).
- **R14. Honest engineering data.** Every constant, table value, and coefficient must carry a source comment (textbook, equation/table, page).

### Mastery, Direction-Agnostic Graphs & Cognitive Fading
- **R15. No magic numbers in learning logic.** Use `phi(n)`, `fib(n)`, `FADING_THRESHOLD`, `snapToFibonacci` from `src/mastery/constants.js`.
- **R16. Variant-aware planning.** Graph reachability calls must receive the active system context: `canSolveTarget(givens, goal, relations, S)`.
- **R17. Respect the mastery lookup contract.** `getDecayedMastery(eq, topic)` reads `eq.masteryId || eq.id` and `eq.baseFormula`. Steps provide `{ id: step.masteryId || step.eqId || step.id, masteryId: step.masteryId, baseFormula: step.baseFormula }`.
- **R18. Direction-Agnostic Problem Solving.** Topics define only the physical network (`varsDb`, `relations`, `createSystem`). There are zero hardcoded starting points, fixed givens, or static goal lists. An engineer must be able to solve problems forwards, backwards, or diagonally.
- **R19. Dynamic Per-Problem Deliverables & Cognitive Fading.** Problem goals are determined dynamically at runtime based on topological distance and mastery level. Mastered intermediate steps ($\text{mastery} \ge \text{FADING THRESHOLD}$) are omitted from statement prompts and **literally hidden** from the interactive solution flow, while the active problem's final deliverable always remains active.

### Derivations & Unit Layout (The Visual Invariants)
- **R20. Textbook derivations.**
  - **Native Degree Symbol:** Native `°` only, never `\circ`, `^\circ`, or `^{\circ}`.
  - **Zero Leading Space in `\text{}`:** Use `\text{kW}`, never `\text{ kW}`.
  - **Strict `\,` Spacing:** Numbers and variables are separated from units by LaTeX thin spacing (`5{,}186{,}450 \, \cfrac{\cancel{\text{Btu}}}{\cancel{\text{h}}}`), including inside compound fraction bars. Zero commas before physical units.
  - **Universal Percent Mirroring:** Percentages render strictly as `\text{\%}` (and `\cancel{\text{\%}}` when cancelled), mirroring all other text units. Never write raw `%`.
  - **Additive Factoring:** Like-unit sums and differences factor outside parentheses: $(a \pm b) \, \text{unit}$ (e.g. $(10 + 20) \, \text{kW}$). Mismatched terms convert in-situ to the target unit before evaluation—even inside nested denominators such as $\frac{1}{1/k_1 + 1/k_2}$.
  - **In-Situ Conversion Default:** Unit conversions are placed immediately adjacent to the token being converted by default.
  - **The Hard 4-Row Fraction Tower Cap:** A compound fraction tower may not exceed 4 vertical visual rows (2 in numerator, 2 in denominator). Stepping down to the main line ($A/B \cdot E/F$) or switching deeply nested unit ratios (`fDepth >= 2`) to inline `\tfrac` occurs *only* to enforce the 4-row cap, or for single-dimension Option B scaling that preserves an intact compound target unit.
  - **The "Omit 1" Rule:** Never write a coefficient of $1$ before a unit token ($\cfrac{\text{kg}}{2.20462 \, \cancel{\text{lb}}}$, $\cfrac{\text{kW}}{1{,}000 \, \cancel{\text{W}}}$, $\cfrac{\cancel{\text{h}}}{60 \, \cancel{\text{min}}}$).
  - **Scalar Multiplier Placement:** Scalar multipliers sit outside the unit fraction: $60 \, \cfrac{\cancel{\text{h}}}{\cancel{\text{min}}}$, never $\cfrac{60 \, \cancel{\text{h}}}{\cancel{\text{min}}}$.
  - **Cancellation Hierarchy:** Wrap the highest possible AST node with `\cancel{}`. Whole-unit cancellation ($\cancel{\cfrac{\text{kJ}}{\text{s}}}$) is canonical; component-level cancellation ($\cfrac{\cancel{\text{Btu}}}{\cancel{\text{h}}}$) is used only as fallback when components cancel in separate places.
- **R21. Specific Energy Notation.** Quantities per unit mass ($\text{kJ/kg}$, $\text{Btu/lb}$) strictly use lowercase symbols ($q_A, q_R, w_{net}, h$), reserving uppercase ($Q, W, P$) for total energy or power.
- **R22. Number formatting.** Measured values pass through `fmtSigFig` (up to 6 significant figures, max 3 decimal places). Unit conversion scale factors pass through `fmtScale` (6 significant figures without the 3-decimal-place truncation cap).

### AI Collaboration, Testing & Process
- **R23. Never make a red test invisible.** No renaming, skipping, narrowing globs, loosening tolerances, or deleting tests. Fix the root cause.
- **R24. Bug fixes ship with regression tests.** Tests that touch mastery use real mastery functions (`recordAttempt`, `getDecayedMastery`), not constant stubs.
- **R25. Report honestly.** State files changed, test output, and anything that could not be verified.
- **R26. Brainstorm mode by default.** Always operate in brainstorm mode first (and whenever a question is asked) to analyze root causes, establish mathematical invariants, and verify alignment. Never write code or emit diffs until given explicit permission to proceed.
- **R27. Canonical diff syntax & tipping point.** When authorized to write code, the diff syntax (`<<<< ==== >>>>`) is the canonical method, strictly when the resulting diff is shorter in character count than returning the drop-in replacement. Documentation (`README.md`) edits are always returned as full drop-in replacements.

### 4.2 `varsDb` Entry (incl. Indexed `_i` Templates)

"T_2": {
  name: "Temperature after Compression",
  symbol: "T_2",
  dimension: DIMENSIONS.TEMPERATURE,
  unit: "K",

- `dimension`: One of `DIMENSIONS.*` (`LENGTH`, `FORCE`, `STIFFNESS`, `PRESSURE`, `STRESS`, `MODULUS`, `POWER`, `TEMPERATURE`, `TIME`, `FREQUENCY`, `ACCELERATION`, etc.).

- `nameAliases` / `symbolAliases`: Pools sampled at runtime for statement variety while derivations maintain canonical notation.
- `isConstant`: Tags fundamental physical constants (e.g. $c_{pa}$, $c_{pw}$, $E$, $g$) for recall-mode omission at high mastery.
- `isInteger`: Answer checking uses $\pm 0.5$ tolerance and the neatness pass rounds to integer.
- `isAbstract`: Tags category parents (e.g. `D_P`, `T`); exempt from orphan audits.

### 4.3 Unified Relation Specification (`defineRelation` / `compileRelations`)

All relations are constructed through `defineRelation` and flattened with `compileRelations`. `nodes` is derived automatically from the AST, and `canonicalTex` is auto-generated from the AST when omitted:

```javascript
export const MY_RELATIONS = compileRelations([
  // 1. Standard equation (nodes & canonicalTex auto-derived)
  defineRelation({
    id: "eq_<topic>_power_flow",
    name: "Power from Mass Flow and Work",
    canonicalTex: "P = \\dot{m} w_{net}",
    formula: "P = m_flow * w_net"
  }),

  // 2. Multi-"=" chain (expands into distinct binary DAG edges)
  defineRelation({
    id: "eq_<topic>_lengths",
    name: "Spring Length Relations",
    formula: "FL = SL + delta_solid = OL + delta"
  }),

  // 3. Continued / nested pattern operators (sum, prod, chain) & indexed _i laws
  defineRelation({
    id: "eq_<topic>_series_k",
    name: "Equivalent Stiffness (Series)",
    formula: "1 / k_T = sum(1 / k_i)"          // Nested e.g. "k_T = 1 / sum(1 / k_i)" also supported
  }),
  defineRelation({
    id: "eq_<topic>_series_force",
    name: "Force Equality (Series)",
    formula: "F_T = chain(F_i)"               // Renders F_T = F_1 = F_2 = ... = F_n; shares masteryId
  }),
  defineRelation({
    id: "eq_<topic>_element_rate",
    name: "Individual Spring Rate",
    formula: "F_i = k_i * delta_i"            // Instantiates i=1..n edges sharing one masteryId
  }),

  // 4. SI / US variants and empirical requiredUnits
  defineRelation({
    id: "eq_<topic>_bending_stress",
    name: "Tooth Bending Stress",
    formula: "s = W_t / (b * m * Y)",
    variants: {
      SI: { canonicalTex: "s = \\frac{F_t}{b m Y}", formula: "s = W_t / (b * m * Y)" },
      US: { canonicalTex: "s = \\frac{F_t P_d}{b Y}", formula: "s = (W_t * P_d) / (b * Y)" }
    }
  })
]);
```

### 4.4 `createSystem({ sys })` Contract

```javascript
export function createMyTopicSystem({ sys = "SI" } = {}) {
  const isUS = sys === "US";

  const m_flow = isUS ? 50.0 : 25.0;        // lb/s vs kg/s
  const w_net = isUS ? 120.0 : 280.0;       // Btu/lb vs kJ/kg
  const P = m_flow * w_net;                 // Pure hand arithmetic

  const _names = {};
  for (const [k, meta] of Object.entries(MY_TOPIC_VARS_DB)) {
    _names[k] = meta.nameAliases?.[0] || meta.name;
  }

  const _u = isUS
    ? { m_flow: "lb/s", w_net: "Btu/lb", P: "Btu/s" }
    : { m_flow: "kg/s", w_net: "kJ/kg", P: "kW" };

  return { sys, m_flow, w_net, P, _names, _u };
}
```

## 5. Formula Language (`formula`)

The parser grammar enforces standard infix mathematical notation:
- Numbers, canonical variable keys, operators `+ - * / ^` (with right-associative `^` and unary `-`), parentheses.
- Functions: `sin cos tan asin acos atan sqrt exp ln log abs`, plus continued constructors `sum(expr)`, `prod(expr)`, `chain(expr)`, and constants `pi`, `e`.
- Angles inside `sin`, `cos`, `tan` accept any expression (e.g. `sin(2 * phi)`, `tan(alpha)`) because all angles convert uniformly to coherent SI radians (`rad`) in pure-physics mode.
- Chained equalities (`A = B = C` or `A = chain(B_i)`) are allowed in `defineRelation` and expand cleanly into binary AST relations.

| Allowed | Forbidden | Failure Mode | Rule |
|---|---|---|---|
| `2 * a` | `2a` | Parser rejects trailing tokens (`null`) | Always include explicit multiplication `*`. |
| `P_sat` | `P_{sat}` | **Silent Truncation** (truncates to `P_`) | No LaTeX braces in variable keys or formulas. |
| `x^(k - 1)` | `x^k-1` | Evaluates as $(x^k) - 1$ | Always parenthesise complex exponents. |
| `a / (b * c)` | `a / b / c` | Left-associative: $(a/b)/c$ | Always parenthesise compound denominators. |
| `sin(2 * phi)` | — | Supported natively via coherent `rad` base | Arbitrary angle expressions evaluate accurately. |

## 6. Physical Units & Dimensional Architecture

### 6.1 The AffineScale Invariant Model
All physical evaluation rests on branchless affine transformations and 7D dimensional vectors:
$$\phi_u(x) = s_u \cdot (x + o_u)$$
$$\phi_u^{-1}(y) = \frac{y}{s_u} - o_u$$

- **Coherent Base Magnitudes:** Every physical state has an invariant representation in coherent SI units ($\text{m}, \text{s}, \text{kg}, \text{N}, \text{K}, \text{W}, \text{J}, \text{rad}, \text{Hz}$).
- **Linear Units ($o_u = 0$):** Multiplicative scale only ($y = s_u \cdot x$).
- **Affine Units ($o_u \neq 0$):** Exact offsets ($\pm 273.15$, $\pm 459.67$). The absolute companion unit is resolved mathematically by matching scale $s$ in thermal dimension $\Theta$ where $o = 0$:
  - $^\circ\text{F}$ ($s = 5/9, o = 459.67$) $\implies ^\circ\text{R}$ ($s = 5/9, o = 0$)
  - $^\circ\text{C}$ ($s = 1.0, o = 273.15$) $\implies \text{K}$ ($s = 1.0, o = 0$)
- **Scale Parity ($|\Delta| < 10^{-6}$):** If two units in a ratio share 7D dimensions and $|s_A - s_B| < 10^{-6}$ (e.g. $\text{kW}$ vs $\text{kJ/s}$, $\text{N/mm}$ vs $\text{kN/m}$, $\text{Hz}$ vs $1/\text{s}$), they are physically equal.
- **Empirical Target Scaling:** When a relation specifies `requiredUnits`, the formula evaluates in those required units; the result then scales to `targetUnit` via `numericConversionFactor = scale(reqTargetU) / scale(targetUnit)`.

### 6.2 `scales.js`: Single Source of Truth & 6-Sig-Fig Precision
`src/cas/units/scales.js` computes all conversion scalars directly from `BASE_UNITS` and `PREFIXES` in `definitions.js` using `fmtScale` (6 significant figures without the 3-decimal truncation cap):
- **Single-Dimension Scaling (Option B):** When an expression evaluates to a single dimension differing from `targetUnit` (e.g. $\text{hp} \to \text{ft}\cdot\text{lb/s}$), `scales.js` preserves the target unit intact on the main line.
- **Compound Ratio Decomposition:** Compound unit conversions (e.g. $\frac{\text{lb/min}}{\text{Btu/h}} \to \frac{\text{kg}}{\text{kWh}}$) and reciprocal Frequency $\leftrightarrow$ Time pairs ($1\,\text{Hz} \cdot 1\,\text{s} = 1$) decompose into atomic factor-label steps applied in-situ.
- **AST-Driven Mismatches:** Ratios with differing unit scales (e.g. $\text{Btu/h}$ vs $\text{Btu/min}$) extract the exact differing atomic pair from the unit AST.

### 6.3 Factor-Label Derivations & The 4-Row Cap
Derivations default to **in-situ conversions** placed directly beside the converted token, including inside nested additive expressions (such as $\frac{1}{1/k_1 + 1/k_2}$).

A compound fraction has 4 vertical visual row slots:
$$\cfrac{\text{Row 1}}{\text{Row 2}} \Bigg/ \cfrac{\text{Row 3}}{\text{Row 4}}$$

- **Stepping Down / Inline Sub-Units:** Stepping down to the main line or switching unit sub-fractions at fraction depth $\ge 2$ to inline `\tfrac` enforces the 4-row cap automatically.

### 6.4 Multi-System Sampling & Coupled State Coherence
- **Problem Distribution:** 40% Pure SI / 40% Pure US / 20% MIXED.
- **Coupled Variable Coherence:** State deliverables of the same fluid or coupled mechanical set share an identical unit when requested as outputs; boundary givens roll freely to test multi-unit fluency.
- **Significant Figure Retention:** Conversions to alternative units must retain $\ge 3$ significant figures under the 3-decimal-place limit ($|v| \ge 0.100$).
- **Compound Decomposition:** Decomposed unit options are sampled with probability inversely proportional to token count: $w_i = 1/t_i$.

## 7. Direction-Agnostic Synthesis & Natural Cognitive Fading

### 7.1 Direction-Agnostic Graph Unrolling
Topics have no fixed starting or ending variables. The generator treats every relation network as an undirected bipartite graph:
- `generateAutonomousProblem` prioritizes the learner's weakest relations and samples any valid variable as the dynamic deliverable (`goalVar`).
- `determineGraphDistance(mastery)` scales chain length monotonically via Fibonacci hops (`0` $\to$ `1` $\to$ `2` $\to$ `3` $\to$ `5` $\to$ `8` $\to$ `21`) as mastery increases.
- `solveDAG` resolves paths forwards, backwards, or diagonally from the minimal pruned givens.

### 7.2 Cognitive Fading Step Hiding
- **Beginner ($< \text{FADING THRESHOLD} \approx 76.4\%$):** Short 0–1 hop problems; every intermediate step is actively prompted.
- **Advanced ($\ge \text{FADING THRESHOLD}$):** Multi-hop interconnected chains; mastered intermediate steps are **literally hidden** from the solution steps and omitted from statement prompts.
- **Ancestry Propagation:** Submitting a correct active step reinforces all its faded ancestor equations (deduplicated by `masteryId`). Failing or revealing an active step penalizes faded ancestors so scaffolding reappears on future problems.

## 8. Derivations & UI Typesetting

- **Canonical Cancellation Hierarchy:** Whole-unit cancellation (`\cancel{\cfrac{\text{kJ}}{\text{s}}}`) is canonical; component-level cancellation is fallback when sub-tokens cancel separately.
- **Omit 1:** Write $\cfrac{\text{kg}}{2.20462 \, \cancel{\text{lb}}}$, never $\cfrac{1 \, \text{kg}}{\dots}$.
- **Scalars Outside:** Write $60 \, \cfrac{\cancel{\text{h}}}{\cancel{\text{min}}}$, never $\cfrac{60 \, \cancel{\text{h}}}{\cancel{\text{min}}}$.
- **Percent Symbol:** Always `\text{\%}` (and `\cancel{\text{\%}}`), mirrored across all unit pipelines.
- **Zero-Comma Spacing:** Units are separated from numbers by LaTeX thin spacing (`\,`), never literal commas.
- **Additive Unit Grouping:** Like-unit sums and differences factor outside parentheses: $(a \pm b) \, \text{unit}$.
- **Perceptual Mastery Gradient:** Formula mastery bars interpolate continuously in Oklab color space from danger red (`oklab(0.63 0.22 0.12)`) through warm amber (`oklab(0.75 0.06 0.16)`) to emerald green (`oklab(0.72 -0.17 0.14)`), sorted ascending.

## 9. Authoring Guide & Fractal Structure for Topics & Subjects

Topics follow a **fractal, self-similar directory structure** under `src/topics/<subject>/`:
- **Branch folders** (contain 2+ subtopics/sub-branches): contain `index.js` (aggregating child topic maps upward) and, whenever children overlap, a `common/` folder (`index.js`, `schema.js`, `relations.js`, and optional `system_helper.js`).
- **Leaf topic folders**: strictly contain `index.js`, `schema.js`, `relations.js`, and `system.js`, importing shared material upward through `../common/index.js`.

```text
src/topics/<subject>/<topic_branch>/
├── index.js           # Aggregates child topics upward
├── common/            # Extracted when 2+ children share schemas/relations
│   ├── index.js       # Barrel re-exporting schema, relations, helpers
│   ├── schema.js      # Shared varsDb definitions
│   └── relations.js   # Shared eq_base_* relations via compileRelations
└── <leaf_topic>/
    ├── index.js       # Topic manifest (or countRange + instantiate(n) for variadic topics)
    ├── schema.js      # Composes ../common/index.js + topic-specific vars
    ├── relations.js   # Composes ../common/index.js + topic-specific defineRelation specs
    └── system.js      # Independent ground truth generator ({ sys } => SystemState)
```
