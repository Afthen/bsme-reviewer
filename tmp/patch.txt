/**
 * src/topics/machine_design/springs/helical_coil/system.js
 * Independent ground-truth generator for helical coil springs.
 */

import { HELICAL_COIL_VARS_DB } from "./schema.js";

export function createHelicalCoilSystem({ sys = "SI" } = {}) {
  const isUS = sys === "US";

  // End connection: standard Squared & Ground (n_dead = 2, k_end = 2)
  const n_dead = 2;
  const k_end = 2;

  let d_w, C, D_m, R_m, D_o, D_i, p, n_e, n_a, SL, FL, delta_solid, alpha, L_wire;
  let K_w, K_b, K_s, K_c, G, k, F, delta, OL, S_f, SPE;
  let F_1, F_2, dF, delta_1, delta_2, ddelta, OL_1, OL_2;
  let _u;

  if (isUS) {
    d_w = 0.25; // in
    C = 8.0;
    D_m = C * d_w; // 2.0 in
    R_m = D_m / 2; // 1.0 in
    D_o = D_m + d_w; // 2.25 in
    D_i = D_m - d_w; // 1.75 in
    p = 0.6; // in
    n_e = 10.0;
    n_a = n_e + n_dead; // 12.0
    SL = (n_e + k_end) * d_w; // 3.0 in
    FL = n_e * p + k_end * d_w; // 6.5 in
    delta_solid = FL - SL; // 3.5 in
    alpha = Math.atan(p / (Math.PI * D_m)) * (180 / Math.PI); // deg
    L_wire = Math.PI * D_m * n_e; // 62.8319 in

    K_w = (4 * C - 1) / (4 * C - 4) + 0.615 / C; // ~1.184018
    K_b = (4 * C + 2) / (4 * C - 3); // ~1.172414
    K_s = 1 + 1 / (2 * C); // 1.0625
    K_c = K_w / K_s; // ~1.11437

    G = 11.5e6; // psi
    k = (G * d_w) / (8 * Math.pow(C, 3) * n_e); // ~70.19043 lb/in
    F = 100.0; // lb
    delta = F / k; // ~1.4247 in
    OL = FL - delta; // ~5.0753 in
    S_f = (8 * K_w * F * C) / (Math.PI * Math.pow(d_w, 2)); // ~38590.2 psi

/**
 * src/topics/machine_design/springs/systems/common/system_helper.js
 * Shared ground truth builder for Series and Parallel spring systems.
 * Eliminates duplicate arithmetic logic across series/parallel system generators.
 */

export function createSpringSystemGroundTruth({ sys = "SI", isParallel = false, count = 3 } = {}) {
  const isUS = sys === "US";

  // Individual spring rates
  const k_rates = isUS
    ? [20.0, 30.0, 40.0, 50.0, 60.0].slice(0, count) // lb/in
    : [1.0, 1.5, 2.0, 2.5, 3.0].slice(0, count);     // N/mm

  let k_T, F_T, delta_T;
  const F_arr = [];
  const delta_arr = [];

  if (isParallel) {
    k_T = k_rates.reduce((sum, ki) => sum + ki, 0);
    delta_T = isUS ? 1.5 : 25.0; // in or mm
    F_T = k_T * delta_T;

    for (let i = 0; i < count; i++) {
      delta_arr[i] = delta_T;
      F_arr[i] = k_rates[i] * delta_T;
    }
  } else {
    const sumRecip = k_rates.reduce((sum, ki) => sum + 1 / ki, 0);
    k_T = 1 / sumRecip;
    F_T = isUS ? 120.0 : 500.0; // lb or N
    delta_T = F_T / k_T;

    for (let i = 0; i < count; i++) {
      F_arr[i] = F_T;
      delta_arr[i] = F_T / k_rates[i];
    }
  }

  const S = { sys, F_T, delta_T, k_T };
  const _u = {
    F_T: isUS ? "lb" : "N",
    delta_T: isUS ? "in" : "mm",
    k_T: isUS ? "lb/in" : "N/mm"
  };

  for (let i = 1; i <= count; i++) {
    const ki = k_rates[i - 1];
    const fi = F_arr[i - 1];
    const di = delta_arr[i - 1];

    S[`k_${i}`] = ki;
    S[`F_${i}`] = fi;
    S[`delta_${i}`] = di;

    _u[`k_${i}`] = isUS ? "lb/in" : "N/mm";
    _u[`F_${i}`] = isUS ? "lb" : "N";
    _u[`delta_${i}`] = isUS ? "in" : "mm";
  }

  return { S, _u };
}

    // Problem 4 exact values: 6-leaf steel, 13 in long, 375 lb, delta = 1.25 in, S = 50,000 psi
    n_leaves = 6;
    L = 13.0; // in
    F = 375.0; // lb
    delta = 1.25; // in
    S = 50000.0; // psi
    E = 30e6; // psi (steel)

    // delta = S*L^2 / (t*E) => t = S*L^2 / (delta*E)
    t = (S * Math.pow(L, 2)) / (delta * E); // 0.22533 in (~0.225 in)
    // S = 6*F*L / (n*b*t^2) => b = 6*F*L / (n*S*t^2)
    b = (6 * F * L) / (n_leaves * S * Math.pow(t, 2)); // ~1.926 in
    k = F / delta; // 300 lb/in

    _u = {
      F: "lb", delta: "in", S: "psi", L: "in",
      b: "in", t: "in", n_leaves: "dimless", E: "psi", k: "lb/in"
    };
  } else {
    // SI metric counterpart
    n_leaves = 6;
    L = 350.0; // mm
    F = 1800.0; // N
    S = 350.0; // MPa
    E = 207.0; // GPa = 207000 MPa
    delta = 30.0; // mm

    const E_mmpa = E * 1000;
    t = (S * Math.pow(L, 2)) / (delta * E_mmpa); // mm
    b = (6 * F * L) / (n_leaves * S * Math.pow(t, 2)); // mm
    k = F / delta; // N/mm

    _u = {
      F: "N", delta: "mm", S: "MPa", L: "mm",
      b: "mm", t: "mm", n_leaves: "dimless", E: "GPa", k: "N/mm"
    };
  }

  const _names = {};
  for (const [key, meta] of Object.entries(CANTILEVER_LEAF_VARS_DB)) {
    _names[key] = meta.nameAliases?.[0] || meta.name;
  }

  return { sys, F, delta, S, L, b, t, n_leaves, E, k, _names, _u };
}

/**
 * src/topics/machine_design/springs/period_frequency/system.js
 * Ground truth generator for spring & pendulum period/frequency.
 */

import { PERIOD_FREQUENCY_VARS_DB } from "./schema.js";

export function createPeriodFrequencySystem({ sys = "SI" } = {}) {
  const isUS = sys === "US";

  let T_period, f, m, k, L_p, g, _u;

  if (isUS) {
    m = 10.0; // lb (pound_mass)
    k = 50.0; // lb/ft
    g = 32.2; // ft/s^2
    const gc = 32.174049;

    // T = 2*pi * sqrt(m / (k * gc))
    T_period = 2 * Math.PI * Math.sqrt(m / (k * gc)); // ~0.495388 s
    f = 1 / T_period; // ~2.0186 Hz
    L_p = g * Math.pow(T_period / (2 * Math.PI), 2); // ~0.20016 ft

    _u = {
      T_period: "s", f: "Hz", m: "lb", k: "lb/ft", L_p: "ft", g: "ft/s^2"
    };
  } else {
    m = 2.5; // kg
    k = 1000.0; // N/m
    g = 9.81; // m/s^2

    T_period = 2 * Math.PI * Math.sqrt(m / k); // 0.314159 s
    f = 1 / T_period; // 3.1831 Hz
    L_p = g * Math.pow(T_period / (2 * Math.PI), 2); // 0.024525 m

    _u = {
      T_period: "s", f: "Hz", m: "kg", k: "N/m", L_p: "m", g: "m/s^2"
    };
  }

  const _names = {};
  for (const [key, meta] of Object.entries(PERIOD_FREQUENCY_VARS_DB)) {
    _names[key] = meta.nameAliases?.[0] || meta.name;
  }

  return { sys, T_period, f, m, k, L_p, g, _names, _u };
}
